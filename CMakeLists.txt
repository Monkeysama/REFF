# 游戏目录由单一 JSON 生成静态 C++ 表；脚本工具直接读取同一 JSON，避免支持状态在多处漂移。
set(REFF_GAME_CATALOG_FILE "${CMAKE_SOURCE_DIR}/config/game-support.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${REFF_GAME_CATALOG_FILE}")
file(READ "${REFF_GAME_CATALOG_FILE}" REFF_GAME_CATALOG_JSON)
string(JSON REFF_GAME_CATALOG_SCHEMA GET "${REFF_GAME_CATALOG_JSON}" schemaVersion)
if(NOT REFF_GAME_CATALOG_SCHEMA EQUAL 1)
  message(FATAL_ERROR "不支持的游戏目录 schemaVersion：${REFF_GAME_CATALOG_SCHEMA}")
endif()
string(JSON REFF_GAME_CATALOG_COUNT LENGTH "${REFF_GAME_CATALOG_JSON}" games)
if(REFF_GAME_CATALOG_COUNT LESS 1)
  message(FATAL_ERROR "游戏目录不能为空")
endif()
math(EXPR REFF_GAME_CATALOG_LAST "${REFF_GAME_CATALOG_COUNT} - 1")
set(REFF_GAME_CATALOG_ENTRIES "")
set(REFF_GAME_CATALOG_TARGETS "")
foreach(REFF_GAME_INDEX RANGE 0 ${REFF_GAME_CATALOG_LAST})
  foreach(REFF_GAME_FIELD target name status renderer directInputKeyboard rawInputKeyboard ime)
    string(JSON REFF_GAME_${REFF_GAME_FIELD} GET "${REFF_GAME_CATALOG_JSON}" games ${REFF_GAME_INDEX} ${REFF_GAME_FIELD})
  endforeach()
  string(JSON REFF_GAME_EXECUTABLE GET "${REFF_GAME_CATALOG_JSON}" games ${REFF_GAME_INDEX} executables 0)
  if(NOT REFF_GAME_target MATCHES "^[A-Z0-9_]+$")
    message(FATAL_ERROR "游戏目录 target 不合法：${REFF_GAME_target}")
  endif()
  list(FIND REFF_GAME_CATALOG_TARGETS "${REFF_GAME_target}" REFF_GAME_DUPLICATE_INDEX)
  if(NOT REFF_GAME_DUPLICATE_INDEX EQUAL -1)
    message(FATAL_ERROR "游戏目录 target 重复：${REFF_GAME_target}")
  endif()
  list(APPEND REFF_GAME_CATALOG_TARGETS "${REFF_GAME_target}")
  string(JSON REFF_GAME_EXECUTABLE_COUNT LENGTH "${REFF_GAME_CATALOG_JSON}" games ${REFF_GAME_INDEX} executables)
  if(REFF_GAME_EXECUTABLE_COUNT LESS 1)
    message(FATAL_ERROR "游戏目录缺少可执行文件：${REFF_GAME_target}")
  endif()
  math(EXPR REFF_GAME_EXECUTABLE_LAST "${REFF_GAME_EXECUTABLE_COUNT} - 1")
  foreach(REFF_GAME_EXECUTABLE_INDEX RANGE 0 ${REFF_GAME_EXECUTABLE_LAST})
    string(JSON REFF_GAME_EXECUTABLE_ALIAS GET "${REFF_GAME_CATALOG_JSON}" games ${REFF_GAME_INDEX} executables ${REFF_GAME_EXECUTABLE_INDEX})
    if(NOT REFF_GAME_EXECUTABLE_ALIAS MATCHES "^[A-Za-z0-9_.-]+\\.exe$")
      message(FATAL_ERROR "游戏目录可执行文件不合法：${REFF_GAME_EXECUTABLE_ALIAS}")
    endif()
  endforeach()
  if(NOT REFF_GAME_status MATCHES "^(recognized|experimental|verified|disabled)$")
    message(FATAL_ERROR "游戏目录 status 不合法：${REFF_GAME_status}")
  endif()
  if(NOT REFF_GAME_renderer STREQUAL "d3d12")
    message(FATAL_ERROR "本轮游戏目录只允许 d3d12：${REFF_GAME_renderer}")
  endif()
  if(NOT REFF_GAME_directInputKeyboard MATCHES "^(disabled|automatic|required)$" OR
     NOT REFF_GAME_rawInputKeyboard MATCHES "^(disabled|automatic|required)$" OR
     NOT REFF_GAME_ime MATCHES "^(disabled|proxy)$")
    message(FATAL_ERROR "游戏目录输入能力不合法：${REFF_GAME_target}")
  endif()
  string(APPEND REFF_GAME_CATALOG_ENTRIES
    "    GameCatalogEntry{\"${REFF_GAME_target}\", \"${REFF_GAME_name}\", \"${REFF_GAME_EXECUTABLE}\", SupportStatus::${REFF_GAME_status}, RendererBackend::${REFF_GAME_renderer}, CapabilityPolicy::${REFF_GAME_directInputKeyboard}, CapabilityPolicy::${REFF_GAME_rawInputKeyboard}, ImePolicy::${REFF_GAME_ime}},\n")
endforeach()
